function selectCategory(type) { document.getElementById('category-sheet').classList.remove('open'); if (!tempCoords) return; let title, desc; if (type === 'repair') { title = 'Проблема (Ремонт)'; desc = 'Нужен ремонт.'; } if (type === 'eco') { title = 'Экология'; desc = 'Вопрос по экологии.'; } if (type === 'help') { title = 'Нужна помощь'; desc = 'Требуется помощь.'; } if (type === 'security') { title = 'Безопасность'; desc = 'Вопрос безопасности.'; } addMarkerToMap(tempCoords, type, title, desc); // УБРАЛИ ЭТУ СТРОЧКУ: // tg.showAlert("📍 Метка добавлена!"); tempCoords = null; }
